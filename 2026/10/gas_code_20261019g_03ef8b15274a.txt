/**
 * Threads クラウド自動投稿（GAS版）
 * 仮想制作オフィス / 3アカウント自動予約投稿
 *
 * 仕組み:
 *  - Googleスプレッドシートの「投稿キュー」シートを読む。
 *  - 「承認」= TRUE かつ 予定日時が現在時刻を過ぎていて まだ未投稿 の行を投稿する。
 *  - 時間主導トリガー（15分ごと推奨）でこの checkAndPost を回す。
 *  - トークンは Script Properties に入れる（このコードには書かない）。
 *
 * 安全設計:
 *  - 承認チェックが入っていない投稿は絶対に公開しない（オーナー確認ゲート）。
 *  - 一度投稿した行は「投稿済み」にして二重投稿しない。
 *  - 1回の実行で投稿する最大件数を制限（暴走防止）。
 */

/**
 * トリガーから呼ばれるメイン関数。

// 投稿キューを「日ごと → アカウントごと（みつえ先生・縁つむぎ婆・アユ・まゆみ）→ 時刻」の順に並べ直す
// （2026-10-19 オーナー「日にちごとに並べて」「4アカウント分けてほしい 見づらいから」）。
// 並び順のための見えない列（L列「並び順」）に鍵を書いて、その列で並べ替える。もう順番どおりなら何もしない。
// A〜D列はアカウントごとに薄い色を付ける。プレビュー列の =IMAGE("URL") はURLを直接書いているので、並べ替えても壊れない。
var BRAND_ORDER = ['みつえ先生', '縁つむぎ婆', '恋愛の魔女アユ', 'まゆみ'];
var BRAND_COLOR = { 'みつえ先生': '#fff4d6', '縁つむぎ婆': '#fde4e8', '恋愛の魔女アユ': '#f1e4ff', 'まゆみ': '#e2f0ff' };
var SORT_KEY_COL = COL_COUNT + 1;

function queueSortKey(row) {
  var v = row[COL.datetime - 1];
  var d = v instanceof Date ? v : new Date(String(v).replace(/-/g, '/'));
  if (isNaN(d.getTime())) return '9999-99-99|9|99:99';
  var b = BRAND_ORDER.indexOf(String(row[COL.brand - 1]));
  return Utilities.formatDate(d, 'Asia/Tokyo', 'yyyy-MM-dd') + '|' + (b < 0 ? 9 : b) + '|' + Utilities.formatDate(d, 'Asia/Tokyo', 'HH:mm');
}

function sortQueueByDate(sheet) {
  var last = sheet.getLastRow();
  if (last < 2) return false;
  var rows = sheet.getRange(2, 1, last - 1, COL_COUNT).getValues();
  var keys = rows.map(queueSortKey);
  var sorted = true;
  for (var i = 1; i < keys.length; i++) { if (keys[i] < keys[i - 1]) { sorted = false; break; } }
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) return false;
  try {
    if (sheet.getMaxColumns() < SORT_KEY_COL) sheet.insertColumnsAfter(sheet.getMaxColumns(), SORT_KEY_COL - sheet.getMaxColumns());
    if (!sorted) {
      sheet.getRange(1, SORT_KEY_COL).setValue('並び順');
      sheet.getRange(2, SORT_KEY_COL, keys.length, 1).setValues(keys.map(function (k) { return [k]; }));
      sheet.getRange(2, 1, last - 1, SORT_KEY_COL).sort({ column: SORT_KEY_COL, ascending: true });
      sheet.hideColumns(SORT_KEY_COL);
      rows = sheet.getRange(2, 1, last - 1, COL_COUNT).getValues();
    }
    // アカウントごとの色（並べ替えの有無にかかわらず、色が違う行だけ塗り直す）
    var want = rows.map(function (r) { var c = BRAND_COLOR[String(r[COL.brand - 1])] || '#ffffff'; return [c, c, c, c]; });
    var have = sheet.getRange(2, 1, last - 1, 4).getBackgrounds();
    var diff = want.some(function (w, i) { return w[0] !== have[i][0]; });
    if (diff) sheet.getRange(2, 1, last - 1, 4).setBackgrounds(want);
    SpreadsheetApp.flush();
  } finally {
    lock.releaseLock();
  }
  return !sorted;
}

// 1回だけの片付け（2026-10-19 オーナー「10月19日の7時台は消していい」「20、21日分も消していい 作り直すから」）。
// まだ出ていない行（状態が空）だけ消す。投稿済み・エラーの行は消さない。終わったら CLEANUP_20261019_DONE を付けて二度と動かない。
function cleanupOnce20261019(sheet) {
  var props = PropertiesService.getScriptProperties();
  if (props.getProperty('CLEANUP_20261019B_DONE')) return 0;
  var last = sheet.getLastRow();
  if (last < 2) return 0;
  var rows = sheet.getRange(2, 1, last - 1, COL_COUNT).getValues();
  var del = [];
  rows.forEach(function (r, i) {
    if (String(r[COL.status - 1]).trim()) return;
    var v = r[COL.datetime - 1];
    var d = v instanceof Date ? v : new Date(String(v).replace(/-/g, '/'));
    if (isNaN(d.getTime())) return;
    var day = Utilities.formatDate(d, 'Asia/Tokyo', 'yyyy-MM-dd');
    var hour = Number(Utilities.formatDate(d, 'Asia/Tokyo', 'H'));
    if (day === '2026-10-19' && hour === 9) del.push(i + 2);  // 2回目（10/19 16時 オーナー「9時台も消していい」）
  });
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) return 0;
  try {
    for (var k = del.length - 1; k >= 0; k--) sheet.deleteRow(del[k]);
    SpreadsheetApp.flush();
  } finally {
    lock.releaseLock();
  }
  props.setProperty('CLEANUP_20261019B_DONE', Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd HH:mm') + ' ' + del.length + '行');
  return del.length;
}

// 1回だけの承認（2026-10-19 オーナー「確認したから、全部承認チェックしていいよ」・10/20と10/21の作り直した40本）。
// まだ出ていない行（状態が空）の承認にチェックを入れる。終わったら APPROVE_20261019_DONE を付けて二度と動かない。
function approveOnce20261019(sheet) {
  var props = PropertiesService.getScriptProperties();
  if (props.getProperty('APPROVE_20261019_DONE')) return 0;
  var last = sheet.getLastRow();
  if (last < 2) return 0;
  var rows = sheet.getRange(2, 1, last - 1, COL_COUNT).getValues();
  var n = 0;
  rows.forEach(function (r, i) {
    if (String(r[COL.status - 1]).trim() || r[COL.approved - 1] === true) return;
    var v = r[COL.datetime - 1];
    var d = v instanceof Date ? v : new Date(String(v).replace(/-/g, '/'));
    if (isNaN(d.getTime())) return;
    var day = Utilities.formatDate(d, 'Asia/Tokyo', 'yyyy-MM-dd');
    if (day === '2026-10-20' || day === '2026-10-21') { sheet.getRange(i + 2, COL.approved).setValue(true); n++; }
  });
  SpreadsheetApp.flush();
  props.setProperty('APPROVE_20261019_DONE', Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd HH:mm') + ' ' + n + '行');
  return n;
}

// 前日までの行を「投稿キュー_過去分」タブへ移す（2026-10-19 オーナー「18日までの欄消していいよ」）。
// 消さずに移すので記録は残る。1日1回（Script Properties の ARCHIVE_DONE_DATE）。並べ替えの後に呼ぶ（古い行が上に固まっている前提）。
var ARCHIVE_SHEET = '投稿キュー_過去分';

function archivePastRows(sheet) {
  var props = PropertiesService.getScriptProperties();
  var today = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd');
  if (props.getProperty('ARCHIVE_DONE_DATE') === today) return 0;
  var last = sheet.getLastRow();
  if (last < 2) return 0;
  var times = sheet.getRange(2, COL.datetime, last - 1, 1).getValues();
  var n = 0;
  for (var i = 0; i < times.length; i++) {
    var v = times[i][0];
    var d = v instanceof Date ? v : new Date(String(v).replace(/-/g, '/'));
    if (isNaN(d.getTime())) break;
    if (Utilities.formatDate(d, 'Asia/Tokyo', 'yyyy-MM-dd') >= today) break;
    n++;
  }
  if (n > 0) {
    var lock = LockService.getScriptLock();
    if (!lock.tryLock(20000)) return 0;
    try {
      var ss = SpreadsheetApp.getActive();
      var arch = ss.getSheetByName(ARCHIVE_SHEET);
      if (!arch) {
        arch = ss.insertSheet(ARCHIVE_SHEET, ss.getSheets().length);
        sheet.getRange(1, 1, 1, sheet.getLastColumn()).copyTo(arch.getRange(1, 1));
        arch.setFrozenRows(1);
      }
      var cols = sheet.getLastColumn();
      sheet.getRange(2, 1, n, cols).copyTo(arch.getRange(arch.getLastRow() + 1, 1));
      sheet.deleteRows(2, n);
      SpreadsheetApp.flush();
    } finally {
      lock.releaseLock();
    }
  }
  props.setProperty('ARCHIVE_DONE_DATE', today);
  return n;
}

function checkAndPost() {
  // インスタのリール（2026-10-16 追加）: 同じ15分おきのトリガーで、承認済み・時刻を過ぎたリールを出す
  try { publishApprovedReels(); } catch (e) { console.log('リールの処理でエラー: ' + e); }
  var ss = SpreadsheetApp.getActive();
  // 「投稿キュー」タブがあればそれを、無ければ最初のタブを使う。
  var sheet = ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0];
  if (!sheet) {
    throw new Error('シートが見つかりません。');
  }

  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return; // ヘッダのみ

  ensureColumns(sheet);
  // 日ごと→アカウントごと→時刻の順に並べる（2026-10-19 オーナー「日にちごとに」「4アカウント分けて」）
  try { approveOnce20261019(sheet); } catch (e) { console.log('承認でエラー: ' + e); }
  try { cleanupOnce20261019(sheet); } catch (e) { console.log('片付けでエラー: ' + e); }
  try { sortQueueByDate(sheet); } catch (e) { console.log('並べ替えでエラー: ' + e); }
  try { archivePastRows(sheet); } catch (e) { console.log('過去分への移動でエラー: ' + e); }
  lastRow = sheet.getLastRow();
  if (lastRow < 2) return;
  var startedAt = Date.now();
  var now = new Date();
  var values = sheet.getRange(2, 1, lastRow - 1, COL_COUNT).getValues();
  var postedCount = 0;

  for (var i = 0; i < values.length; i++) {
    if (postedCount >= MAX_POSTS_PER_RUN) break;
    if (Date.now() - startedAt > RUN_BUDGET_MS) break; // 残りは次のトリガーで投稿する

    var row = values[i];
    var rowIndex = i + 2; // 実シート行

    var brand = String(row[COL.brand - 1]).trim();
    var text = String(row[COL.text - 1]);
    var approved = row[COL.approved - 1] === true;
    var status = String(row[COL.status - 1]).trim();
    var scheduled = row[COL.datetime - 1];

    // 既に処理済み・未承認・本文なしはスキップ（一部エラー＝1本目は公開済みなので再投稿しない）
    if (status === '投稿済み' || status === 'エラー' || status === '一部エラー') continue;
    if (!approved) continue;
    if (!brand || !text) continue;

    // 予定時刻がまだ来ていない場合はスキップ
    var scheduledDate = (scheduled instanceof Date) ? scheduled : new Date(scheduled);
    if (isNaN(scheduledDate.getTime())) {
      writeResult(sheet, rowIndex, 'エラー', '予定日時が読めません: ' + scheduled, now);
      continue;
    }
    if (scheduledDate.getTime() > now.getTime()) continue;

    var imageUrls = splitImages(row[COL.images - 1]);
    var follows = splitThread(row[COL.thread - 1]);

    // 投稿実行（1本目が失敗したらエラー。1本目が出た後の続きで失敗したら一部エラー）
    var ids = [];
    try {
      checkLength('本文', text);
      follows.forEach(function (f, n) { checkLength('続き' + (n + 1), f); });
      if (imageUrls.length > CAROUSEL_MAX) {
        throw new Error('画像は' + CAROUSEL_MAX + '枚までです（' + imageUrls.length + '枚）');
      }
      imageUrls.forEach(function (url) {
        if (!/^https:\/\//.test(url)) throw new Error('画像URLは https:// で始まる公開URLにしてください: ' + url);
      });
      var token = getToken(brand);
      ids.push(publishMainPost(token, text, imageUrls));
      for (var f = 0; f < follows.length; f++) {
        Utilities.sleep(3000);
        ids.push(publishContainer(token, { media_type: 'TEXT', text: follows[f], reply_to_id: ids[ids.length - 1] }));
      }
      writeResult(sheet, rowIndex, '投稿済み', ids.join(' / '), now);
      postedCount++;
    } catch (e) {
      var message = String(e && e.message ? e.message : e);
      if (ids.length) {
        writeResult(sheet, rowIndex, '一部エラー', '公開済み: ' + ids.join(' / ') + ' ／ 続きで失敗: ' + message, now);
        postedCount++;
      } else {
        writeResult(sheet, rowIndex, 'エラー', message, now);
      }
    }
  }
}

/**
 * 予定日時を「yyyy-MM-dd HH:mm」の文字列にそろえる（上書き対象の突き合わせ用）。
 */
function scheduleKey(value) {
  var d = (value instanceof Date) ? value : new Date(value);
  if (isNaN(d.getTime())) return String(value).trim();
  return Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm');
}

function lastRowForReplace(sheet) {
  return Math.max(sheet.getLastRow(), 1);
}

/**
 * 指定した日付（yyyy-MM-dd）の「未承認・未投稿」の行を空にする（2026-09-18追加）。
 * 承認済みの行と、状態が入っている行（投稿済み・エラー）には触れない。
 * 行は削除せず中身だけ消すので、空いた行は doPost の追加でそのまま再利用される。
 */
function clearPendingRows(sheet, dates) {
  var targets = {};
  [].concat(dates).forEach(function (d) { targets[String(d).trim()] = true; });
  var lastRow = lastRowForReplace(sheet);
  if (lastRow < 2) return 0;

  var values = sheet.getRange(2, 1, lastRow - 1, COL_COUNT).getValues();
  var cleared = 0;
  values.forEach(function (row, i) {
    if (!row[COL.brand - 1]) return;
    if (row[COL.approved - 1] === true) return;
    if (String(row[COL.status - 1]).trim()) return;
    if (!targets[scheduleKey(row[COL.datetime - 1]).slice(0, 10)]) return;
    var rowIndex = i + 2;
    sheet.getRange(rowIndex, 1, 1, COL_COUNT).clearContent();
    sheet.getRange(rowIndex, COL.approved).insertCheckboxes();
    sheet.getRange(rowIndex, COL.approved).setValue(false);
    sheet.setRowHeight(rowIndex, 21);
    cleared++;
  });
  return cleared;
}

function checkLength(label, value) {
  if (value.length > TEXT_LIMIT) {
    throw new Error(label + 'が' + TEXT_LIMIT + '文字を超えています（' + value.length + '字）');
  }
}

function splitImages(cell) {
  return String(cell || '').split(/[\n,]/).map(function (s) { return s.trim(); }).filter(String);
}

function splitThread(cell) {
  return String(cell || '').split(THREAD_SEPARATOR).map(function (s) { return s.trim(); }).filter(String);
}

/**
 * シートの列が11列に足りなければ増やし、I〜K列の見出しが空なら入れる（3ブランドの既存行はそのまま）。
 */
function ensureColumns(sheet) {
  if (sheet.getMaxColumns() < COL_COUNT) {
    sheet.insertColumnsAfter(sheet.getMaxColumns(), COL_COUNT - sheet.getMaxColumns());
  }
  var header = sheet.getRange(1, 1, 1, COL_COUNT).getValues()[0];
  for (var c = COL.images - 1; c < COL_COUNT; c++) {
    if (!header[c]) sheet.getRange(1, c + 1).setValue(HEADERS[c]);
  }
}

/**
 * 画像をスプレッドシートのエディタから手動で準備したい時用（見出しの追加だけ行う）。
 */
function setupImageColumns() {
  var ss = SpreadsheetApp.getActive();
  ensureColumns(ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0]);
}

/**
 * ブランドに対応するトークンを Script Properties から取得。
 */
function getToken(brand) {
  var key = TOKEN_KEYS[brand];
  if (!key) throw new Error('未知のブランド: ' + brand);
  var token = PropertiesService.getScriptProperties().getProperty(key);
  if (!token) throw new Error(key + ' が Script Properties に設定されていません。');
  return token;
}

/**
 * URLが動画かどうか。拡張子だけで見る（I列に動画のURLを書けば動画投稿になる）。
 * 2026-09-30 追加。列を増やさずに済ませるため、画像URLの列をそのまま使う。
 */
function isVideoUrl(url) {
  return /\.(mp4|mov)(\?|$)/i.test(String(url || '').split('#')[0]);
}

/**
 * 1本目を公開する。
 *   何もなし = 文字だけ / 動画1本 = 動画投稿 / 画像1枚 = 画像投稿 / 画像2枚以上 = カルーセル。
 * threads_api/threads_workflow.py の publish_item と同じ手順（PC版で実投稿確認済み）。
 * 戻り値: 公開後の投稿ID
 */
function publishMainPost(token, text, imageUrls) {
  if (!imageUrls.length) {
    return publishContainer(token, { media_type: 'TEXT', text: text });
  }

  // 動画（2026-09-30 追加）。Threads APIは media_type=VIDEO + video_url。
  // 仕様: MP4/MOV・H264/HEVC・最大1920px・最大5分・最大1GB・moov atomが先頭（faststart）。
  var videos = imageUrls.filter(isVideoUrl);
  if (videos.length) {
    if (imageUrls.length > 1) {
      throw new Error('動画は1本だけにしてください（動画と画像を混ぜた投稿には未対応）。');
    }
    // 動画は取り込みに時間がかかるので、待ち時間を長めに取る
    return publishContainer(token, { media_type: 'VIDEO', video_url: videos[0], text: text }, 200000);
  }

  if (imageUrls.length === 1) {
    return publishContainer(token, { media_type: 'IMAGE', image_url: imageUrls[0], text: text });
  }
  var children = imageUrls.map(function (url) {
    var childId = createContainer(token, { media_type: 'IMAGE', image_url: url, is_carousel_item: 'true' });
    waitForContainerReady(childId, token);
    return childId;
  });
  return publishContainer(token, { media_type: 'CAROUSEL', children: children.join(','), text: text });
}

 * 互換用: 文字だけの投稿（以前の関数名）。

function publishThreadsPost(token, text) {

}

/**
 * Threads公式API: コンテナ作成 → 準備完了を待つ → 公開。戻り値: 公開後の投稿ID
 */
function publishContainer(token, params, maxWaitMs) {
  var creationId = createContainer(token, params);
  waitForContainerReady(creationId, token, maxWaitMs);
  var pubBody = graphPost(token, 'me/threads_publish', { creation_id: creationId });
  if (!pubBody.id) throw new Error('公開失敗: ' + JSON.stringify(pubBody));
  return pubBody.id;
}

function createContainer(token, params) {
  var body = graphPost(token, 'me/threads', params);
  if (!body.id) throw new Error('コンテナ作成失敗: ' + JSON.stringify(body));
  return body.id;
}

function graphPost(token, path, params) {
  var payload = { access_token: token };
  Object.keys(params).forEach(function (k) { payload[k] = params[k]; });
  var res = UrlFetchApp.fetch(GRAPH_BASE + '/' + path, { method: 'post', muteHttpExceptions: true, payload: payload });
  var body = JSON.parse(res.getContentText() || '{}');
  if (res.getResponseCode() >= 300) throw new Error(path + ' 失敗: ' + res.getContentText());
  return body;
}

/**
 * コンテナ(creationId)の処理が完了する(status=FINISHED)まで待つ。
 *
 * 原因（2026-08-11調査）: コンテナ作成は非同期処理のため、作成直後にpublishすると
 * 「error_subcode 4279009 / Media Not Found」で失敗することがある。
 * 2026-09-15: 取得フィールドを status_code → status に修正（APIの項目名は status。
 * 以前は項目が返らず毎回30秒待ってから公開していた）。画像は取り込みに時間がかかるため最大120秒待つ。
 */
function waitForContainerReady(creationId, token, maxWaitMs) {
  var MAX_WAIT_MS = maxWaitMs || 120000;
  var POLL_INTERVAL_MS = 3000;
  var waited = 0;

  Utilities.sleep(2000);
  waited += 2000;

  while (waited < MAX_WAIT_MS) {
    var statusUrl = GRAPH_BASE + '/' + creationId + '?fields=status,error_message&access_token=' + encodeURIComponent(token);
    var res = UrlFetchApp.fetch(statusUrl, { muteHttpExceptions: true });
    var body = JSON.parse(res.getContentText() || '{}');
    var state = body.status || body.status_code;

    if (state === 'FINISHED') return;
    if (state === 'ERROR' || state === 'EXPIRED') {
      throw new Error('コンテナ処理失敗: ' + state + ' ' + (body.error_message || ''));
    }
    Utilities.sleep(POLL_INTERVAL_MS);
    waited += POLL_INTERVAL_MS;
  }
  throw new Error('投稿の準備が' + (MAX_WAIT_MS / 1000) + '秒で終わりませんでした');
}

/**
 * 公開URLのファイルを取ってきて、Googleドライブに保存する。
 * 同じ名前のファイルが既にあれば置き換える（何度走らせても増えない）。
 * 戻り値: { name, id, url, size }
 */
function saveUrlToDrive(url, filename, folderId) {
  if (!url) throw new Error('url がありません。');
  var name = filename || String(url).split('/').pop().split('?')[0];
  name = decodeURIComponent(name);

  var res = UrlFetchApp.fetch(url, { muteHttpExceptions: true, followRedirects: true });
  if (res.getResponseCode() >= 300) {
    throw new Error('取得できませんでした(' + res.getResponseCode() + '): ' + url);
  }
  var blob = res.getBlob().setName(name);

  var folder = folderId ? DriveApp.getFolderById(folderId) : DriveApp.getRootFolder();

  // 同じ名前の古いファイルはゴミ箱へ（差し替え）
  var existing = folder.getFilesByName(name);
  while (existing.hasNext()) {
    existing.next().setTrashed(true);
  }

  var file = folder.createFile(blob);
  return {
    name: file.getName(),
    id: file.getId(),
    url: file.getUrl(),
    size: file.getSize()
  };
}

/**
 * 書き込み専用のWeb App窓口。
 * ClaudeがHTTP POSTで投稿予定を送り、行として追加する。
 *
 * リクエストの形:
 * {
 *   "secret": "合言葉",
 *   "posts": [
 *     {"brand": "みつえ先生", "handle": "mitsu.esensei_uranai", "datetime": "2026-07-09 7:30", "text": "..."},
 *     {"brand": "まゆみ", "handle": "mayumi_lovetarot", "datetime": "2026-09-17 9:00", "text": "...",
 *      "images": ["https://raw.githubusercontent.com/..."], "thread": ["2本目", "3本目"]},
 *     ...
 *   ]
 * }
 *
 * 安全設計:
 *  - secret が SCRIPT_WRITE_SECRET と一致しないと何もしない。
 *  - 承認（E列）は必ずFALSEで追加する。ここではオーナーの確認ゲートを絶対に飛ばさない。
 *  - 状態・結果・投稿実行時刻は空のまま追加する（checkAndPostが後で埋める）。
 */
function doPost(e) {
  var response = { ok: false, added: 0, errors: [] };
  try {
    var body = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActive();
    var sheet = ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0];
    ensureColumns(sheet);

    if (body.action === 'sync') {
      syncFromPc(body);
      response.ok = true;
      return ContentService.createTextOutput(JSON.stringify(response)).setMimeType(ContentService.MimeType.JSON);
    }

    var expected = PropertiesService.getScriptProperties().getProperty('SCRIPT_WRITE_SECRET');
    if (!expected) {
      throw new Error('SCRIPT_WRITE_SECRET が Script Properties に設定されていません。');
    }
    if (!body.secret || body.secret !== expected) {
      throw new Error('合言葉が一致しません。');
    }

    // 返信キュー（2026-10-15 追加）: PCが集めたコメントとAIの返信案を「返信キュー」タブに入れる
    if (body.action === 'replyStatus') {
      response.status = replyStatus();
      response.ok = true;
      return ContentService.createTextOutput(JSON.stringify(response)).setMimeType(ContentService.MimeType.JSON);
    }
    if (body.action === 'replies') {
      response.added = addReplyRows(body.rows || []);
      response.ok = true;
      return ContentService.createTextOutput(JSON.stringify(response)).setMimeType(ContentService.MimeType.JSON);
    }

    // 見出しの修復と在庫の一括承認（2026-10-17 追加）。ブラウザのキー操作は不安定なので、スプシの変更はここから行う
    if (body.action === 'restoreHeaders') {
      var hdr = sheet.getRange(1, 1, 1, HEADERS.length).getValues()[0];
      var fixed = [];
      HEADERS.forEach(function (h, i) { if (String(hdr[i]).trim() === '') { sheet.getRange(1, i + 1).setValue(h); fixed.push(h); } });
      response.fixed = fixed;
      response.ok = true;
      return ContentService.createTextOutput(JSON.stringify(response)).setMimeType(ContentService.MimeType.JSON);
    }
    if (body.action === 'approveBank') {  // オーナーが「全部チェックして」と言った時だけ使う
      var bsh = SpreadsheetApp.getActive().getSheetByName(BANK_SHEET);
      var n = bsh ? bsh.getLastRow() - 1 : 0;
      if (n > 0) bsh.getRange(2, BK.approved, n, 1).setValue(true);
      response.approved = n;
      response.ok = true;
      return ContentService.createTextOutput(JSON.stringify(response)).setMimeType(ContentService.MimeType.JSON);
    }
    // 返事の在庫（2026-10-17 追加）: PCから在庫を入れる（承認FALSE）
    if (body.action === 'bank') {
      response.added = addBankRows(body.rows || []);
      response.ok = true;
      return ContentService.createTextOutput(JSON.stringify(response)).setMimeType(ContentService.MimeType.JSON);
    }
    // 予約の重複チェック（2026-10-16 追加・オーナー指示「重複は絶対に防ぐ」）。毎朝 daily_report.py が読む
    if (body.action === 'queueCheck') {
      response.check = checkQueue(sheet);
      response.ok = true;
      return ContentService.createTextOutput(JSON.stringify(response)).setMimeType(ContentService.MimeType.JSON);
    }
    // インスタのリール（2026-10-16 追加）
    if (body.action === 'reels') {
      response.added = addReelRows(body.rows || []);
      response.ok = true;
      return ContentService.createTextOutput(JSON.stringify(response)).setMimeType(ContentService.MimeType.JSON);
    }
    if (body.action === 'syncIgfb') {
      syncIgfb(body);
      response.ok = true;
      return ContentService.createTextOutput(JSON.stringify(response)).setMimeType(ContentService.MimeType.JSON);
    }

    if (body.action === 'clear') {
      response.cleared = clearPendingRows(sheet, body.dates || []);
      response.ok = true;
      return ContentService.createTextOutput(JSON.stringify(response)).setMimeType(ContentService.MimeType.JSON);
    }

    // 公開URLのファイルをGoogleドライブへ保存する（2026-09-30 追加）
    //
    // なぜ必要か: Instagramのリールは、アプリから手で投稿しないと
    // Instagramの音源（曲）を付けられない。だから動画はスマホに届いていればよい。
    // Claude側からドライブへ直接アップロードする手段が無い（コネクタはファイルの中身を
    // 会話に載せる方式で、動画のサイズでは通らない）。
    // GASはオーナーのアカウントで動くので、URLを渡せばドライブに置ける。
    if (body.action === 'saveToDrive') {
      response.saved = saveUrlToDrive(body.url, body.filename, body.folderId);
      response.ok = true;
      return ContentService.createTextOutput(JSON.stringify(response)).setMimeType(ContentService.MimeType.JSON);
    }

    var posts = body.posts || [];
    if (!posts.length) {
      throw new Error('posts が空です。');
    }

    // ブランド(A列)が空の行を2行目から集める。
    // これで、既にチェックボックスが設定済みの空き行を優先的に再利用する
    // （appendRowだけだと末尾に追記され、チェックボックス範囲外になることがあるため）。
    var lastRow = Math.max(sheet.getLastRow(), 1);
    var emptyRows = [];
    if (lastRow >= 2) {
      var brandColumn = sheet.getRange(2, COL.brand, lastRow - 1, 1).getValues();
      for (var r = 0; r < brandColumn.length; r++) {
        if (!brandColumn[r][0]) {
          emptyRows.push(r + 2); // 実シート行番号
        }
      }
    }
    var nextNewRow = lastRow + 1;

    // 同じブランド・同じ予定日時の「未承認・未投稿」の行があれば、その行を上書きする（2026-09-17 追加。
    // 送り直しで同じ予定が二重に並ぶのを防ぐ）。承認済みの行は上書きしない。
    var replaceRows = {};
    if (lastRowForReplace(sheet) >= 2) {
      var existing = sheet.getRange(2, 1, lastRowForReplace(sheet) - 1, COL_COUNT).getValues();
      existing.forEach(function (row, i) {
        if (!row[COL.brand - 1] || row[COL.approved - 1] === true || String(row[COL.status - 1]).trim()) return;
        replaceRows[String(row[COL.brand - 1]).trim() + '|' + scheduleKey(row[COL.datetime - 1])] = i + 2;
      });
    }

    posts.forEach(function (post) {
      if (!post.brand || !post.handle || !post.datetime || !post.text) {
        response.errors.push('必須項目が足りない行をスキップしました: ' + JSON.stringify(post));
        return;
      }
      if (!TOKEN_KEYS[post.brand]) {
        response.errors.push('未知のブランドの行をスキップしました: ' + post.brand);
        return;
      }
      var images = [].concat(post.images || []).filter(String);
      var thread = [].concat(post.thread || []).filter(String);
      var preview = images.length ? '=IMAGE("' + String(images[0]).replace(/"/g, '') + '")' : '';
      var replaceKey = post.brand + '|' + scheduleKey(post.datetime);
      var targetRow = replaceRows[replaceKey] || (emptyRows.length > 0 ? emptyRows.shift() : nextNewRow++);
      delete replaceRows[replaceKey];
      // 承認(E)は常にFALSEで書き込む。状態・結果・投稿実行時刻(F,G,H)は空にする。
      sheet.getRange(targetRow, 1, 1, COL_COUNT).setValues(
        [[post.brand, post.handle, post.datetime, post.text, false, '', '', '',
          images.join('\n'), preview, thread.join('\n---\n')]]
      );
      sheet.getRange(targetRow, COL.approved).insertCheckboxes(); // 末尾に追記した行でもチェックボックスにする
      sheet.getRange(targetRow, COL.approved).setValue(false);
      if (preview) sheet.setRowHeight(targetRow, 120);
      response.added++;
    });

    response.ok = true;
  } catch (err) {
    response.errors.push(String(err && err.message ? err.message : err));
  }
  return ContentService
    .createTextOutput(JSON.stringify(response))
    .setMimeType(ContentService.MimeType.JSON);
}

/**
 * PCからトークン（と合言葉）を届けてもらう窓口（2026-09-16）。手で貼り付けなくて済むようにする。
 * {"action":"sync","brand":"まゆみ","token":"...","secret":"..."(任意)}
 * 安全: 届いたトークンで Threads の /me を呼び、ユーザー名が BRAND_HANDLES と一致した時だけ保存する。
 * つまり、そのアカウントの有効なトークンを既に持っている人しか書き換えられない。ログにトークンは出さない。
 */
function syncFromPc(body) {
  var brand = String(body.brand || '');
  var key = TOKEN_KEYS[brand];
  var handle = BRAND_HANDLES[brand];
  if (!key || !handle) throw new Error('未知のブランド: ' + brand);
  var token = String(body.token || '');
  if (!token) throw new Error('token が空です');
  var res = UrlFetchApp.fetch(GRAPH_BASE + '/me?fields=username&access_token=' + encodeURIComponent(token), { muteHttpExceptions: true });
  var me = JSON.parse(res.getContentText() || '{}');
  if (res.getResponseCode() >= 300 || me.username !== handle) {
    throw new Error('トークンを確認できませんでした（' + brand + '）');
  }
  var props = PropertiesService.getScriptProperties();
  props.setProperty(key, token);
  if (body.secret) props.setProperty('SCRIPT_WRITE_SECRET', String(body.secret));
}

/**
 * 60日トークンを期限前に延長する（2026-09-15追加）。週1回の時間主導トリガーで回す。
 * 発行から24時間以上たった、まだ期限内のトークンだけ延長できる。期限切れは再発行が必要。
 * 延長した新しいトークンは Script Properties に上書き保存する（ログにトークンは出さない）。
 */
function refreshTokens() {
  var props = PropertiesService.getScriptProperties();
  Object.keys(TOKEN_KEYS).forEach(function (brand) {
    var key = TOKEN_KEYS[brand];
    var token = props.getProperty(key);
    if (!token) {
      console.log(brand + ': 未設定のためスキップ');
      return;
    }
    var res = UrlFetchApp.fetch('https://graph.threads.net/refresh_access_token?grant_type=th_refresh_token&access_token=' +
      encodeURIComponent(token), { muteHttpExceptions: true });
    var body = JSON.parse(res.getContentText() || '{}');
    if (res.getResponseCode() < 300 && body.access_token) {
      props.setProperty(key, body.access_token);
      var expires = new Date(Date.now() + Number(body.expires_in || 0) * 1000);
      console.log(brand + ': 延長しました（期限 ' + Utilities.formatDate(expires, Session.getScriptTimeZone(), 'yyyy-MM-dd') + '）');
    } else {
      console.log(brand + ': 延長に失敗 ' + ((body.error && body.error.message) || res.getContentText()));
    }
  });
}

// PCが毎朝読む: 止まっているアカウントと、今日送った件数
function replyStatus() {
  var out = {};
  var sh = SpreadsheetApp.getActive().getSheetByName(REPLY_SHEET);
  var today = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd');
  Object.keys(TOKEN_KEYS).forEach(function (b) { out[b] = { sentToday: 0, waiting: 0, errors: 0, pausedUntil: replyPausedUntil(b) ? Utilities.formatDate(new Date(replyPausedUntil(b)), 'Asia/Tokyo', 'MM/dd HH:mm') : '' }; });
  if (sh && sh.getLastRow() >= 2) {
    sh.getRange(2, 1, sh.getLastRow() - 1, REPLY_HEADERS.length).getValues().forEach(function (r) {
      var b = r[RC.brand - 1];

      if (!st && r[RC.approved - 1] !== true) out[b].waiting++;
      if (st.indexOf('エラー') === 0) out[b].errors++;
    });
  }
  return out;
}

    added++;
  });

}

function sendApprovedReplies() {

    if (r[RC.approved - 1] !== true || String(r[RC.status - 1]).trim() || done[brand]) continue;
    if (replyPausedUntil(brand)) continue; // 制限で止めている間は送らない
    if ((sentToday[brand] || 0) >= REPLY_DAILY_MAX) continue;
    if (lastSent[brand] && (Date.now() - lastSent[brand].getTime()) < REPLY_GAP_MS) continue;
    var rowIndex = i + 2;
    var now = new Date();
    try {
      var text = String(r[RC.reply - 1]).trim();
      checkLength('返信', text);
      var postedId = publishContainer(getToken(brand), { media_type: 'TEXT', text: text, reply_to_id: String(r[RC.commentId - 1]) });
      sh.getRange(rowIndex, RC.status).setValue('送信済み');
      sh.getRange(rowIndex, RC.result).setValue(String(postedId));
      sh.getRange(rowIndex, RC.sentAt).setValue(now);
      sentToday[brand] = (sentToday[brand] || 0) + 1;
    } catch (err) {
      var msg = String(err);
      if (LIMIT_ERROR.test(msg)) {
        PropertiesService.getScriptProperties().setProperty(pauseKey(brand), String(Date.now() + REPLY_PAUSE_MS));
        sh.getRange(rowIndex, RC.status).setValue('エラー（制限のため24時間停止）');
      } else {
        sh.getRange(rowIndex, RC.status).setValue('エラー');
      }
      sh.getRange(rowIndex, RC.result).setValue(msg.slice(0, 300));
      sh.getRange(rowIndex, RC.sentAt).setValue(now);
    }
    done[brand] = true; // 1回の実行で1アカウント1件まで（5分ごとのトリガーと合わせて間隔があく）
  }
}

function syncIgfb(body) {
  var brand = String(body.brand || '');
  var key = IG_KEYS[brand];
  var handle = BRAND_HANDLES[brand];
  if (!key || !handle) throw new Error('未知のブランド: ' + brand);
  var token = String(body.token || ''), igUser = String(body.igUser || '');
  if (!token || !igUser) throw new Error('token か igUser が空です');
  var res = UrlFetchApp.fetch(IG_GRAPH + '/' + igUser + '?fields=username&access_token=' + encodeURIComponent(token), { muteHttpExceptions: true });
  var me = JSON.parse(res.getContentText() || '{}');
  if (res.getResponseCode() >= 300 || me.username !== handle) throw new Error('インスタの鍵を確認できませんでした（' + brand + '）');
  var props = PropertiesService.getScriptProperties();
  props.setProperty('IGFB_TOKEN_' + key, token);
  props.setProperty('IGFB_IGUSER_' + key, igUser);
}

function igCall(method, path, token, params) {
  var opt = { method: method, muteHttpExceptions: true };
  var url = IG_GRAPH + '/' + path;
  params = params || {};
  params.access_token = token;
  var q = Object.keys(params).map(function (k) { return encodeURIComponent(k) + '=' + encodeURIComponent(params[k]); }).join('&');
  if (method === 'get') { url += '?' + q; } else { opt.payload = q; opt.contentType = 'application/x-www-form-urlencoded'; }
  var res = UrlFetchApp.fetch(url, opt);
  var body = JSON.parse(res.getContentText() || '{}');
  if (res.getResponseCode() >= 300 || body.error) throw new Error('インスタAPI ' + res.getResponseCode() + ': ' + JSON.stringify(body.error || body).slice(0, 300));
  return body;
}

function publishApprovedReels() {
  var sh = SpreadsheetApp.getActive().getSheetByName(REEL_SHEET);
  if (!sh || sh.getLastRow() < 2) return;
  var values = sh.getRange(2, 1, sh.getLastRow() - 1, REEL_HEADERS.length).getValues();
  var props = PropertiesService.getScriptProperties();
  var now = new Date(), done = 0;
  for (var i = 0; i < values.length && done < REEL_MAX_PER_RUN; i++) {
    var r = values[i], row = i + 2;
    var status = String(r[RL.status - 1]).trim();
    if (r[RL.approved - 1] !== true || (status && status !== '処理中')) continue;
    var at = r[RL.at - 1] instanceof Date ? r[RL.at - 1] : new Date(String(r[RL.at - 1]).replace(/-/g, '/'));
    if (isNaN(at.getTime()) || at > now) continue;
    var key = IG_KEYS[r[RL.brand - 1]];
    var token = props.getProperty('IGFB_TOKEN_' + key), uid = props.getProperty('IGFB_IGUSER_' + key);
    try {
      if (!token || !uid) throw new Error('IGFB_TOKEN_' + key + ' が未設定（PCから syncIgfb で届ける）');
      var creationId = status === '処理中' ? String(r[RL.result - 1]) : '';
      if (!creationId) {
        var p = { media_type: 'REELS', video_url: String(r[RL.video - 1]), caption: String(r[RL.caption - 1] || '') };
        // 音量は必ず「1.0」「0.0」の小数で送る。JSON.stringify だと 1.0 が「1」になり、インスタが1%と読んで曲がとても小さくなった（2026-10-18 実測: PCから出した10/16は-17dB、GASの10/17・18は-56dB）
        if (String(r[RL.audioId - 1])) p.audio_configuration = '{"audio_id":"' + String(r[RL.audioId - 1]) + '","audio_volume":1.0,"video_volume":0.0}';
        if (String(r[RL.thumb - 1]) !== '') p.thumb_offset = String(r[RL.thumb - 1]);
        creationId = igCall('post', uid + '/media', token, p).id;
        sh.getRange(row, RL.status).setValue('処理中');
        sh.getRange(row, RL.result).setValue(creationId);
      }
      var ready = false;
      for (var w = 0; w < 8; w++) {  // 最大2分待つ。終わらなければ次の15分で続き
        var st = igCall('get', creationId, token, { fields: 'status_code,status' });
        if (st.status_code === 'FINISHED') { ready = true; break; }
        if (st.status_code === 'ERROR' || st.status_code === 'EXPIRED') throw new Error('動画の処理に失敗: ' + (st.status || st.status_code));
        Utilities.sleep(15000);
      }
      if (!ready) continue;
      var pub = igCall('post', uid + '/media_publish', token, { creation_id: creationId });
      sh.getRange(row, RL.status).setValue('投稿済み');
      sh.getRange(row, RL.result).setValue(String(pub.id));
      sh.getRange(row, RL.postedAt).setValue(now);
      done++;
    } catch (e) {
      sh.getRange(row, RL.status).setValue('エラー');
      sh.getRange(row, RL.result).setValue(String(e).slice(0, 300));
      sh.getRange(row, RL.postedAt).setValue(now);
    }
  }
}

// 予約の重複チェック（2026-10-16 追加）
// これから出る行（状態が空・予定日が今日以降）を数える。
//  counts: 「日付|ブランド」ごとの本数（Threadsは1日5本のはず）
//  sameSlot: 同じ予定日時・同じブランドの行が2つ以上
//  sameText: 同じブランドで本文の先頭60文字（空白を除く）が同じ行（投稿済みの行とも比べる）
// ===================================================================
function checkQueue(sheet) {
  var last = sheet.getLastRow();
  var out = { total: 0, counts: {}, sameSlot: [], sameText: [] };
  if (last < 2) return out;
  var values = sheet.getRange(2, 1, last - 1, COL_COUNT).getValues();
  var today = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd');
  var slots = {}, texts = {}, futureTextKeys = {};
  // 過去分タブに移した投稿の本文も数える（同じ本文を二度出さないため・2026-10-19）
  var arch = SpreadsheetApp.getActive().getSheetByName(ARCHIVE_SHEET);
  if (arch && arch.getLastRow() >= 2) {
    arch.getRange(2, 1, arch.getLastRow() - 1, COL_COUNT).getValues().forEach(function (r) {
      var tk = String(r[COL.brand - 1]) + '|' + String(r[COL.text - 1]).replace(/\s+/g, '').slice(0, 60);
      texts[tk] = (texts[tk] || 0) + 1;
    });
  }
  values.forEach(function (r) {
    var at = r[COL.datetime - 1];
    var key = at instanceof Date ? Utilities.formatDate(at, 'Asia/Tokyo', 'yyyy-MM-dd HH:mm') : String(at).trim();
    var day = key.slice(0, 10);
    var brand = String(r[COL.brand - 1]);
    var tkey = brand + '|' + String(r[COL.text - 1]).replace(/\s+/g, '').slice(0, 60);
    texts[tkey] = (texts[tkey] || 0) + 1;
    var status = String(r[COL.status - 1]).trim();
    if (status || day < today || !brand) return;
    out.total++;
    out.counts[day + '|' + brand] = (out.counts[day + '|' + brand] || 0) + 1;
    var skey = key + '|' + brand;
    slots[skey] = (slots[skey] || 0) + 1;
    futureTextKeys[tkey] = key;
  });
  Object.keys(slots).forEach(function (k) { if (slots[k] > 1) out.sameSlot.push('同じ日時: ' + k + '（' + slots[k] + '行）'); });
  Object.keys(futureTextKeys).forEach(function (k) {
    if (texts[k] > 1) out.sameText.push('同じ本文: ' + futureTextKeys[k] + ' ' + k.split('|')[0] + '「' + k.split('|')[1].slice(0, 20) + '」（' + texts[k] + '行）');
  });
  return out;
}

function autoReplyFromBank() {
  var ss = SpreadsheetApp.getActive();
  var bankSh = ss.getSheetByName(BANK_SHEET);
  if (!bankSh || bankSh.getLastRow() < 2) return;
  var bank = bankSh.getRange(2, 1, bankSh.getLastRow() - 1, BANK_HEADERS.length).getValues();
  var postSh = ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0];
  var now = new Date(), today = Utilities.formatDate(now, 'Asia/Tokyo', 'yyyy-MM-dd');
  var logSh = getSheetWithHeaders(AUTO_LOG_SHEET, AUTO_LOG_HEADERS);
  var logVals = logSh.getLastRow() >= 2 ? logSh.getRange(2, 1, logSh.getLastRow() - 1, AUTO_LOG_HEADERS.length).getValues() : [];

  // 二重に返さない: 記録・返信キューに入っているコメントID
  var done = {}, sentToday = {}, lastSent = {};
  logVals.forEach(function (r) {
    done[String(r[2])] = true;
    var t = r[6] ? new Date(r[6]) : null, b = r[0];
    if (t && String(r[7]) === '送信済み') {
      if (Utilities.formatDate(t, 'Asia/Tokyo', 'yyyy-MM-dd') === today) sentToday[b] = (sentToday[b] || 0) + 1;
      if (!lastSent[b] || t > lastSent[b]) lastSent[b] = t;
    }
  });
  var rq = ss.getSheetByName(REPLY_SHEET);
  if (rq && rq.getLastRow() >= 2) {
    rq.getRange(2, 1, rq.getLastRow() - 1, REPLY_HEADERS.length).getValues().forEach(function (r) {
      done[String(r[RC.commentId - 1])] = true;
      if (String(r[RC.status - 1]) === '送信済み' && r[RC.sentAt - 1]) {
        var t = new Date(r[RC.sentAt - 1]), b = r[RC.brand - 1];
        if (Utilities.formatDate(t, 'Asia/Tokyo', 'yyyy-MM-dd') === today) sentToday[b] = (sentToday[b] || 0) + 1;
        if (!lastSent[b] || t > lastSent[b]) lastSent[b] = t;
      }
    });
  }

  // 対象の投稿（投稿済み・2時間以内・呼びかけあり）
  var targets = {};
  var pv = postSh.getRange(2, 1, postSh.getLastRow() - 1, COL_COUNT).getValues();
  pv.forEach(function (r) {
    if (String(r[COL.status - 1]) !== '投稿済み' || !r[COL.postedAt - 1]) return;
    var pa = r[COL.postedAt - 1];
    var postedAt = pa instanceof Date ? pa : new Date(String(pa).replace(/-/g, '/'));
    if (isNaN(postedAt.getTime()) || now - postedAt > AUTO_WINDOW_MS) return;
    if (String(r[COL.text - 1]).indexOf('置いて') < 0) return;
    var id = String(r[COL.result - 1]).split('/')[0].trim();
    if (!id) return;
    var b = r[COL.brand - 1];
    (targets[b] = targets[b] || []).push(id);
  });

  Object.keys(targets).forEach(function (brand) {
    if (replyPausedUntil(brand)) return;
    if ((sentToday[brand] || 0) >= REPLY_DAILY_MAX) return;
    if (lastSent[brand] && now - lastSent[brand] < REPLY_GAP_MS) return;
    var token;
    try { token = getToken(brand); } catch (e) { return; }
    var handle = BRAND_HANDLES[brand];
    for (var p = 0; p < targets[brand].length; p++) {
      var postId = targets[brand][p], conv;
      try { conv = threadsGet(token, postId + '/conversation', { fields: 'id,text,username,replied_to,is_reply_owned_by_me', reverse: 'false' }).data || []; }
      catch (e) { console.log(brand + ' の会話を取れませんでした: ' + e); continue; }
      var answered = {};
      conv.forEach(function (c) { if (c.is_reply_owned_by_me && c.replied_to) answered[c.replied_to.id] = true; });
      var target = null;
      for (var i = 0; i < conv.length; i++) {
        var c = conv[i];
        if (c.is_reply_owned_by_me || c.username === handle || answered[c.id] || done[c.id]) continue;
        if (!c.replied_to || c.replied_to.id !== postId) continue;  // 投稿への直接のコメントだけ
        var txt = String(c.text || '').trim();
        if (!txt || txt.length > AUTO_MAX_LEN) continue;            // 長い相談は返信キューへ
        target = c; break;
      }
      if (!target) continue;
      var w = String(target.text).trim();
      var kind = w.length <= 4 ? '一言' : '短文';
      var choices = [];
      bank.forEach(function (r, idx) { if (r[BK.brand - 1] === brand && r[BK.kind - 1] === kind && r[BK.approved - 1] === true) choices.push(idx); });
      if (!choices.length) return;  // 承認された在庫が無ければ送らない
      choices.sort(function (x, y) { return (bank[x][BK.used - 1] || 0) - (bank[y][BK.used - 1] || 0) || Math.random() - 0.5; });
      var pick = choices[0];
      var text = String(bank[pick][BK.text - 1]).replace('{w}', w);
      var result = '送信済み';
      try {
        checkLength('返信', text);
        publishContainer(token, { media_type: 'TEXT', text: text, reply_to_id: String(target.id) });
      } catch (e) {
        result = 'エラー: ' + String(e).slice(0, 200);
        if (LIMIT_ERROR.test(String(e))) PropertiesService.getScriptProperties().setProperty(pauseKey(brand), String(Date.now() + REPLY_PAUSE_MS));
      }
      logSh.appendRow([brand, postId, String(target.id), target.username || '', w, text, now, result]);
      bankSh.getRange(pick + 2, BK.used).setValue((bank[pick][BK.used - 1] || 0) + 1);
      bankSh.getRange(pick + 2, BK.lastUsed).setValue(now);
      return;  // 1回に1アカウント1件
    }
  });
}

    var t = r[6] ? new Date(r[6]) : null, b = r[0];
    if (t && String(r[7]) === '送信済み') {
      if (Utilities.formatDate(t, 'Asia/Tokyo', 'yyyy-MM-dd') === today) sentToday[b] = (sentToday[b] || 0) + 1;
      if (!lastSent[b] || t > lastSent[b]) lastSent[b] = t;
    }
  });

  Object.keys(IG_KEYS).forEach(function (brand) {
    var key = IG_KEYS[brand];
    var token = props.getProperty('IGFB_TOKEN_' + key), uid = props.getProperty('IGFB_IGUSER_' + key);
    if (!token || !uid) return;
    var until = Number(props.getProperty(igPauseKey(brand)) || 0);
    if (until && Date.now() < until) return;
    if ((sentToday[brand] || 0) >= IG_REPLY_DAILY_MAX) return;
    if (lastSent[brand] && now - lastSent[brand] < REPLY_GAP_MS) return;
    var hasBank = bank.some(function (r) { return r[BK.brand - 1] === brand && String(r[BK.kind - 1]).indexOf('IG') === 0 && r[BK.approved - 1] === true; });
    if (!hasBank) return;  // 承認された在庫が無ければ何もしない

    var media;
    try { media = igCall('get', uid + '/media', token, { fields: 'id,timestamp,media_product_type,comments_count', limit: 15 }).data || []; }
    catch (e) { console.log(brand + ' のリールを読めませんでした: ' + e); return; }
    var since = now.getTime() - IG_REPLY_WINDOW_DAYS * 24 * 60 * 60 * 1000;
    for (var m = 0; m < media.length; m++) {
      var md = media[m];
      if (md.media_product_type !== 'REELS' || !md.comments_count) continue;
      if (new Date(md.timestamp).getTime() < since) continue;
      var comments;
      try { comments = igCall('get', md.id + '/comments', token, { fields: 'id,text,from,replies{id}', limit: 50 }).data || []; }
      catch (e) { console.log(brand + ' のコメントを読めませんでした: ' + e); continue; }
      var target = null;
      for (var i = 0; i < comments.length; i++) {
        var c = comments[i];
        if (done[c.id]) continue;
        if (c.from && String(c.from.id) === String(uid)) continue;            // 自分のコメント
        if (c.replies && c.replies.data && c.replies.data.length) continue;  // もう返している
        var txt = String(c.text || '').trim();
        if (!txt || txt.length > AUTO_MAX_LEN) continue;                      // 長い相談は手で返す
        target = c; break;
      }
      if (!target) continue;
      var w = String(target.text).trim();
      var kind = w.length <= 4 ? 'IG一言' : 'IG短文';
      var choices = [];
      bank.forEach(function (r, idx) { if (r[BK.brand - 1] === brand && r[BK.kind - 1] === kind && r[BK.approved - 1] === true) choices.push(idx); });
      if (!choices.length) return;
      choices.sort(function (x, y) { return (bank[x][BK.used - 1] || 0) - (bank[y][BK.used - 1] || 0) || Math.random() - 0.5; });
      var pick = choices[0];
      var text = String(bank[pick][BK.text - 1]).replace('{w}', w);
      var result = '送信済み';
      try {
        igCall('post', String(target.id) + '/replies', token, { message: text });
      } catch (e) {
        result = 'エラー: ' + String(e).slice(0, 200);
        if (LIMIT_ERROR.test(String(e))) props.setProperty(igPauseKey(brand), String(Date.now() + REPLY_PAUSE_MS));
      }
      logSh.appendRow([brand, 'IG:' + md.id, String(target.id), (target.from && target.from.username) || '', w, text, now, result]);
      bankSh.getRange(pick + 2, BK.used).setValue((bank[pick][BK.used - 1] || 0) + 1);
      bankSh.getRange(pick + 2, BK.lastUsed).setValue(now);
      bank[pick][BK.used - 1] = (bank[pick][BK.used - 1] || 0) + 1;
      return;  // 1回に1アカウント1件
    }
  });
}
